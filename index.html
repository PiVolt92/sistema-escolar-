// Google Apps Script - Sincronizador Web y Google Sheets function doGet(e) { try { var ss = SpreadsheetApp.getActiveSpreadsheet(); var sheet = ss.getSheetByName("Datos") || ss.getSheets()[0]; var valor = sheet.getRange("A1").getValue(); // Si la celda está vacía, devuelve un objeto vacío if (!valor) valor = "{}"; return ContentService.createTextOutput(valor) .setMimeType(ContentService.MimeType.JSON); } catch (err) { return ContentService.createTextOutput(JSON.stringify({ error: err.toString() })) .setMimeType(ContentService.MimeType.JSON); } } function doPost(e) { try { var ss = SpreadsheetApp.getActiveSpreadsheet(); var sheet = ss.getSheetByName("Datos") || ss.getSheets()[0]; // Recibe los datos enviados desde la web y los guarda en A1 var data = e.postData.contents; sheet.getRange("A1").setValue(data); return ContentService.createTextOutput(JSON.stringify({ status: "ok" })) .setMimeType(ContentService.MimeType.JSON); } catch (err) { return ContentService.createTextOutput(JSON.stringify({ error: err.toString() })) .setMimeType(ContentService.MimeType.JSON); } }
